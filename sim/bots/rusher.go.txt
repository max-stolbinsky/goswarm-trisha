// Rusher: CTF bot that mirrors Maia Warrior — wall the flag, then race
// straight (L-path) at the mirrored enemy-flag estimate and shoot walls.

const (
	Up = iota
	Right
	Down
	Left
)

type Unit struct {
	tx, ty, budget, phase int
	moves, scan, detour   int
	detourDir             int
	horiz                 bool
	dodge                 bool
}

var units = map[int]*Unit{}
var walled [4]bool
var wallTries [4]int
var tick, lastID int
var dx = [4]int{0, 1, 0, -1}
var dy = [4]int{-1, 0, 1, 0}

func abs(v int) int {
	if v < 0 {
		return -v
	}
	return v
}

func me() *Unit {
	id := my_id()
	u, ok := units[id]
	if !ok {
		u = &Unit{}
		units[id] = u
	}
	return u
}

func turnTo(dir int) {
	if (dir-get_dir()+4)%4 == 3 {
		turn_left()
	} else {
		turn_right()
	}
}

func target(u *Unit, x, y int) {
	cx, cy := get_pos()
	u.tx, u.ty = x, y
	u.budget = 3*(abs(x-cx)+abs(y-cy)) + 20
	u.detour = 0
	u.horiz = abs(x-cx) >= abs(y-cy)
}

func near(u *Unit) bool {
	x, y := get_pos()
	return abs(u.tx-x)+abs(u.ty-y) <= 1
}

func toward(ddx, ddy int, horiz bool) int {
	if (horiz && ddx != 0) || ddy == 0 {
		if ddx > 0 {
			return Right
		}
		if ddx < 0 {
			return Left
		}
		return get_dir()
	}
	if ddy > 0 {
		return Down
	}
	return Up
}

func step(u *Unit) {
	u.budget--
	x, y := get_pos()
	ddx, ddy := u.tx-x, u.ty-y
	want := u.detourDir
	if u.detour > 0 {
		u.detour--
	} else {
		horiz := u.horiz
		if u.horiz {
			horiz = ddx != 0
		} else {
			horiz = ddx != 0 && ddy == 0
		}
		want = toward(ddx, ddy, horiz)
	}
	if get_dir() != want {
		turnTo(want)
		return
	}
	what, dist := look_forward()
	if what == "enemy" {
		shoot()
		return
	}
	if what == "built_wall" && dist <= 2 {
		fx, fy := my_flag()
		if abs(x-fx)+abs(y-fy) > 4 {
			shoot()
			return
		}
	}
	if what == "ally" && dist == 1 {
		u.detourDir = (want + 1) % 4
		u.detour = 2
		turnTo(u.detourDir)
		return
	}
	if move_forward() {
		u.moves++
		return
	}
	u.detourDir = toward(ddx, ddy, want == Up || want == Down)
	if u.detourDir == want {
		u.detourDir = (want + 1) % 4
	}
	u.detour = 3
	u.horiz = !u.horiz
}

func makeWarrior() {
	if !build_warrior() {
		turn_right()
	}
}

func builder(u *Unit, what string, dist int) {
	if busy() > 0 {
		return
	}
	if my_units() < 3 {
		makeWarrior()
		return
	}
	fx, fy := my_flag()
	for side := Up; side <= Left; side++ {
		if walled[side] {
			continue
		}
		sx, sy := fx+2*dx[side], fy+2*dy[side]
		g := get_grid_size()
		if sx < 0 || sy < 0 || sx >= g || sy >= g {
			walled[side] = true
			continue
		}
		x, y := get_pos()
		if x != sx || y != sy {
			if u.tx != sx || u.ty != sy {
				target(u, sx, sy)
			}
			if u.budget <= 0 {
				walled[side] = true
			}
			step(u)
			return
		}
		face := (side + 2) % 4
		if get_dir() != face {
			turnTo(face)
			return
		}
		if (what == "built_wall" && dist == 1) || build_wall() || wallTries[side] > 5 {
			walled[side] = true
		}
		wallTries[side]++
		return
	}
	if my_units() < 6 {
		makeWarrior()
		return
	}
	turn_right()
}

func raid(u *Unit, what string, dist int) {
	if what == "built_wall" && dist <= 2 {
		fx, fy := my_flag()
		x, y := get_pos()
		if abs(x-fx)+abs(y-fy) > 4 {
			shoot()
			return
		}
	}
	g := get_grid_size()
	fx, fy := my_flag()
	// Mirror our flag across the board — close to the enemy flag on CTF maps.
	ex, ey := g-1-fx, g-1-fy
	if u.phase == 0 || u.budget <= 0 || near(u) {
		if u.phase == 0 {
			target(u, ex, ey)
			u.phase = 1
		} else if u.phase == 1 {
			// Spiral search around estimate.
			off := 2 + (u.moves % 5)
			target(u, ex+off*(1-2*(my_id()%2)), ey+off*(1-2*((my_id()/2)%2)))
			u.phase = 2
		} else {
			target(u, 2+randInt(g-4), 2+randInt(g-4))
		}
	}
	if u.moves >= 6 {
		u.moves = 0
		u.scan = 3
		turn_right()
		return
	}
	step(u)
}

func loop() {
	if my_id() <= lastID {
		tick++
	}
	lastID = my_id()
	u := me()
	what, dist := look_forward()
	if what == "flag" || what == "enemy_flag" {
		move_forward()
		return
	}
	if my_kind() == "builder" {
		builder(u, what, dist)
		return
	}
	if what == "enemy" {
		shoot()
		return
	}
	if what == "bullet" {
		u.dodge = true
		turn_right()
		return
	}
	if u.dodge {
		u.dodge = false
		move_forward()
		return
	}
	if u.scan > 0 {
		u.scan--
		turn_right()
		return
	}
	if game_mode() == "ctf" {
		raid(u, what, dist)
		return
	}
	// Non-CTF: simple roam/shoot.
	if u.budget <= 0 || near(u) {
		g := get_grid_size()
		target(u, 2+randInt(g-4), 2+randInt(g-4))
	}
	step(u)
}
